rule "Primo VE - Lds17a"
when
  MARC."700" has any "a,b,1" 
  AND MARC."700"."1" match ".+wikidata.+"
then
set TEMP"1" to MARC."700" subfields "1"
set TEMP"2" to MARC."700" subfields "a"
set TEMP"3" to MARC."700" subfields "b"
concatenate with delimiter (TEMP"1",TEMP"2","")
add suffix (TEMP"2","</a><img class=\"wologo\" src=\"https://www.wikidata.org/static/images/icons/wikidatawiki.svg\" alt=\"wikidata\" width=\"25\" />")
add prefix (TEMP"1","<a target=\"_blank\" href=\"")
add suffix (TEMP"1","\">")
concatenate with delimiter (TEMP"1", TEMP"2", "")
create pnx."display"."lds17" with TEMP"1"
end
